Total Matches = COUNTROWS(IPL_Cleaned)
Total Runs = SUM(IPL_Cleaned[Total_Runs])
Total Wickets = SUM(IPL_Cleaned[Total_Wickets])
Total Teams = DISTINCTCOUNT(Team_Wins[Team])
Total Players = DISTINCTCOUNT(Top_Batsmen[Batter])
Average Runs per Match = DIVIDE([Total Runs], [Total Matches])
Average Winning Margin = AVERAGE(MatchData[Result_Margin])
